    23→export function normalizeName(s: string): string {
    24→  return s
    25→    .toLowerCase()
    26→    .normalize('NFKD')
    27→    .replace(/[\u0300-\u036f]/g, '')
    28→    .replace(/[^a-z0-9\u4e00-\u9fff\u3040-\u30ff\uac00-\ud7af]/g, '')
    29→}

    31→export function slugify(s: string): string {
    32→  const base = s
    33→    .toLowerCase()
    34→    .normalize('NFKD')
    35→    .replace(/[\u0300-\u036f]/g, '')
    36→    .replace(/[^a-z0-9]+/g, '-')
    37→    .replace(/^-+|-+$/g, '')
    38→    .slice(0, 80)
    39→  return base || 'aesthetic'
    40→}

    42→export function parseYear(period: string, which: 'start' | 'end'): number | null {
    43→  if (!period) return null
    44→  const p = period.toLowerCase()
    45→  const mCent = p.match(/(\d{1,2})(?:st|nd|rd|th)\s*century/)
    46→  if (mCent) {
    47→    const n = parseInt(mCent[1], 10)
    48→    let base = (n - 1) * 100
    49→    if (which === 'end') base += 99
    50→    if (p.includes('bc') || p.includes('bce')) base = -base
    51→    return base
    52→  }
    53→  const nums = [...p.matchAll(/(\d{3,4})\s*(bce|bc)?/g)].map((m) => {
    54→    const v = parseInt(m[1], 10)
    55→    return m[2] ? -v : v
    56→  })
    57→  if (nums.length === 0) return null
    58→  if (which === 'start') return nums[0]
    59→  return nums.length > 1 ? nums[nums.length - 1] : nums[0]
    60→}

    70→// Simple global throttle so concurrent workers respect API rate limits.
    71→// Raised from 3.5s after sustained 429 storms; 6s + per-worker jitter keeps
    72→// throughput near the sustainable ceiling without wasting requests.
    73→const MIN_GAP_MS = 6000
    74→let lastRequestStart = 0
    75→async function throttle() {
    76→  for (;;) {
    77→    const now = Date.now()
    78→    const wait = lastRequestStart + MIN_GAP_MS - now
    79→    if (wait <= 0) {
    80→      lastRequestStart = now
    81→      return
    82→    }
    83→    await sleep(Math.min(wait, 1000))
    84→  }
    85→}

    87→// Circuit breaker: when the API sustains 429s, fail fast for a cooldown
    88→// window instead of burning retries (and discovery-batch attempts) against
    89→// a closed endpoint. The worker treats this as a transient condition.
    90→let rateLimitedUntil = 0
    91→export function rateLimitedNow(): boolean {
    92→  return Date.now() < rateLimitedUntil
    93→}
    94→export function isRateLimitedError(e: unknown): boolean {
    95→  const msg = String((e as Error)?.message ?? '')
    96→  return msg.includes('429') || msg.toLowerCase().includes('too many') || msg.includes('rate-limited (cooldown)')
    97→}

    99→export async function llmJSON(system: string, user: string, retries = 5): Promise<any> {
   100→  if (Date.now() < rateLimitedUntil) {
   101→    throw new Error(`rate-limited (cooldown ${Math.ceil((rateLimitedUntil - Date.now()) / 1000)}s)`)
   102→  }
   103→  let lastErr: Error | null = null
   104→  for (let i = 1; i <= retries; i++) {
   105→    try {
   106→      await throttle()
   107→      const z = await zai()
   108→      const completion = await z.chat.completions.create({
   109→        messages: [
   110→          { role: 'assistant', content: system },
   111→          { role: 'user', content: user },
   112→        ],
   113→        thinking: { type: 'disabled' },
   114→      })
   115→      const raw = completion.choices[0]?.message?.content ?? ''
   116→      const json = extractJSON(raw)
   117→      if (json === null) throw new Error('no JSON in response')
   118→      rateLimitedUntil = 0
   119→      return json
   120→    } catch (e: any) {
   121→      lastErr = e
   122→      const msg = String(e?.message ?? '')
   123→      if (i < retries) {
   124→        const is429 = msg.includes('429') || msg.toLowerCase().includes('too many')
   125→        await sleep(is429 ? Math.min(20000 * i, 60000) : 2000 * i)
   126→      }
   127→    }
   128→  }
   129→  if (isRateLimitedError(lastErr)) {
   130→    rateLimitedUntil = Date.now() + 180_000
   131→    logRate('sustained 429s — cooling down 3 minutes')
   132→  }
   133→  throw lastErr ?? new Error('LLM failed')
   134→}

   136→let lastRateLog = 0
   137→function logRate(msg: string) {
   138→  const now = Date.now()
   139→  if (now - lastRateLog > 60_000) {
   140→    lastRateLog = now
   141→    console.log(new Date().toISOString().slice(11, 19), msg)
   142→  }
   143→}

   145→export function extractJSON(raw: string): any {
   146→  if (!raw) return null
   147→  let t = raw.trim()
   148→  t = t.replace(/^```(?:json)?/i, '').replace(/```$/i, '').trim()
   149→  const firstArr = t.indexOf('[')
   150→  const firstObj = t.indexOf('{')
   151→  try {
   152→    if (firstArr !== -1 && (firstObj === -1 || firstArr < firstObj)) {
   153→      const last = t.lastIndexOf(']')
   154→      return JSON.parse(t.slice(firstArr, last + 1))
   155→    }
   156→    if (firstObj !== -1) {
   157→      const last = t.lastIndexOf('}')
   158→      return JSON.parse(t.slice(firstObj, last + 1))
   159→    }
   160→  } catch {
   161→    // fall through to truncated-JSON repair
   162→  }
   163→  // Repair: responses truncated or with per-element errors (output token limit).
   164→  // Recover each complete top-level object from the array independently.
   165→  if (firstArr !== -1) {
   166→    let depth = 0
   167→    let inStr = false
   168→    let esc = false
   169→    const spans: Array<[number, number]> = []
   170→    let objStart = -1
   171→    for (let i = firstArr; i < t.length; i++) {
   172→      const c = t[i]
   173→      if (inStr) {
   174→        if (esc) esc = false
   175→        else if (c === '\\') esc = true
   176→        else if (c === '"') inStr = false
   177→        continue
   178→      }
   179→      if (c === '"') inStr = true
   180→      else if (c === '{') {
   181→        if (depth === 0) objStart = i
   182→        depth++
   183→      } else if (c === '}') {
   184→        depth--
   185→        if (depth === 0 && objStart !== -1) {
   186→          spans.push([objStart, i])
   187→          objStart = -1
   188→        }
   189→      }
   190→    }
   191→    const salvaged: any[] = []
   192→    for (const [a, b] of spans) {
   193→      try {
   194→        salvaged.push(JSON.parse(t.slice(a, b + 1)))
   195→      } catch {}
   196→    }
   197→    if (salvaged.length > 0) return salvaged
   198→  }
   199→  return null
   200→}

   237→function clampInt(v: any, min = 0, max = 100): number {
   238→  const n = Math.round(Number(v))
   239→  if (!Number.isFinite(n)) return 50
   240→  return Math.max(min, Math.min(max, n))

   243→function strArr(v: any, max = 10): string[] {
   244→  if (!Array.isArray(v)) return []
   245→  return v
   246→    .filter((x) => typeof x === 'string' && x.trim().length > 0 && x.trim().length < 200)
   247→    .map((x) => x.trim())
   248→    .slice(0, max)

   251→function strVal(v: any, max = 1200): string {
   252→  if (typeof v !== 'string') return ''
   253→  return v.trim().slice(0, max)

   256→function objVal(v: any): Record<string, string> {
   257→  if (!v || typeof v !== 'object' || Array.isArray(v)) return {}
   258→  const out: Record<string, string> = {}
   259→  for (const [k, val] of Object.entries(v as any)) {
   260→    if (typeof val === 'string' && val.trim()) out[k.slice(0, 40)] = val.trim().slice(0, 500)
   261→    else if (typeof val === 'number') out[k.slice(0, 40)] = String(val)
   262→  }
   263→  return out
   264→}

   270→function colorArr(v: any): Array<{ hex: string; name: string }> {
   271→  if (!Array.isArray(v)) return []
   272→  const out: Array<{ hex: string; name: string }> = []
   273→  for (const c of v) {
   274→    let hex = ''
   275→    let name = ''
   276→    if (typeof c === 'string') {
   277→      hex = c
   278→    } else if (c && typeof c === 'object') {
   279→      hex = String(c.h ?? c.hex ?? '')
   280→      name = String(c.n ?? c.name ?? '').slice(0, 60)
   281→    }
   282→    hex = hex.trim()
   283→    if (/^#[0-9a-fA-F]{6}$/.test(hex)) out.push({ hex: hex.toLowerCase(), name })
   284→    else if (/^#[0-9a-fA-F]{3}$/.test(hex)) {
   285→      out.push({ hex: '#' + hex.slice(1).split('').map((ch) => ch + ch).join(''), name })
   286→    }
   287→    if (out.length >= 7) break
   288→  }
   289→  return out
   290→}

   292→function sourceArr(v: any): Array<{ name: string; url: string; tier: string }> {
   293→  if (!Array.isArray(v)) return []
   294→  const out: Array<{ name: string; url: string; tier: string }> = []
   295→  for (const s of v) {
   296→    if (!s) continue
   297→    let name = ''
   298→    let url = ''
   299→    let tier = 'C'
   300→    if (typeof s === 'string') {
   301→      name = s.slice(0, 160)
   302→    } else if (typeof s === 'object') {
   303→      name = String(s.n ?? s.name ?? '').slice(0, 160)
   304→      url = String(s.u ?? s.url ?? '').slice(0, 300)
   305→      tier = String(s.t ?? s.tier ?? 'C').toUpperCase().slice(0, 1)
   306→    }
   307→    if (!name && !url) continue
   308→    if (!'ABCD'.includes(tier)) tier = 'C'
   309→    if (url && !/^https?:\/\//.test(url)) url = url.startsWith('www.') ? 'https://' + url : ''
   310→    if (!name) name = url.replace(/^https?:\/\/(www\.)?/, '').slice(0, 80)
   311→    out.push({ name, url, tier })
   312→    if (out.length >= 4) break
   313→  }
   314→  return out
   315→}

   353→export function validateEntry(raw: RawEntry, defaultCategory: string): ValidEntry | { error: string } {
   354→  const name = strVal(raw.n ?? raw.name, 120)
   355→  if (!name || name.length < 2) return { error: 'missing name' }
   356→  const desc = strVal(raw.desc ?? raw.description, 2500)
   357→  if (desc.length < 40) return { error: `description too short for ${name}` }
   358→
   359→  const establishmentRaw = strVal(raw.e ?? raw.establishment, 40)
   360→  const establishment = ESTABLISHMENTS.has(establishmentRaw) ? establishmentRaw : classifyEstablishment(defaultCategory)
   361→
   362→  const dnaIn = raw.dna && typeof raw.dna === 'object' ? raw.dna : {}
   363→  const dnaAxes: Record<string, number> = {}
   364→  for (const [short, full] of DNA_KEYS) {
   365→    const v = dnaIn[short] ?? dnaIn[full]
   366→    dnaAxes[full] = clampInt(v)
   367→  }
   368→  const emoIn = raw.emo && typeof raw.emo === 'object' ? raw.emo : {}
   369→  const emotionProfile: Record<string, number> = {}
   370→  for (const k of EMO_KEYS) {
   371→    const v = emoIn[k]
   372→    emotionProfile[k] = clampInt(v)
   373→  }
   374→
   375→  const period = strVal(raw.p ?? raw.period, 80)
   376→  const periodStart = /[–—-]/.test(period) ? period.split(/[–—-]/)[0].trim() : period
   377→  const periodEnd = /[–—-]/.test(period) ? (period.split(/[–—-]/)[1]?.trim() ?? '') : ''
   378→
   379→  const sources = sourceArr(raw.src ?? raw.sources)
   380→
   381→  return {
   382→    name,
   383→    aliases: strArr(raw.a ?? raw.aliases, 8),
   384→    category: strVal(raw.cat, 60) || defaultCategory,
   385→    subcategory: strVal(raw.c ?? raw.subcategory, 120),
   386→    establishment,
   387→    summary: strVal(raw.sum ?? raw.summary, 400) || desc.slice(0, 180),
   388→    description: desc,
   389→    origin: strVal(raw.o ?? raw.origin, 120),
   390→    geography: strVal(raw.g ?? raw.geography, 160),
   391→    periodStart,
   392→    periodEnd,
   393→    era: strVal(raw.era, 60),
   394→    colors: colorArr(raw.col ?? raw.colors),
   395→    materials: strArr(raw.m ?? raw.materials, 10),
   396→    textures: strArr(raw.tx ?? raw.textures, 8),
   397→    objects: strArr(raw.obj ?? raw.objects, 10),
   398→    keyExamples: strArr(raw.ex ?? raw.keyExamples, 6),
   399→    typography: objVal(raw.typ ?? raw.typography),
   400→    visualDNA: objVal(raw.vd ?? raw.visualDNA),
   401→    recipe: objVal(raw.rec ?? raw.recipe),
   402→    dnaAxes,
   403→    emotionProfile,
   404→    sources,
   405→    tags: strArr(raw.tg ?? raw.tags, 10).map((t) => t.toLowerCase().slice(0, 40)),
   406→    popularity: clampInt(raw.pop ?? raw.popularity ?? 50),
   407→    isNiche: raw.niche === true || clampInt(raw.pop ?? 50) < 30,
   408→    influences: strArr(raw.inf ?? raw.influences, 6),
   409→    related: strArr(raw.rel ?? raw.related, 8),
   410→    confidence: sources.length >= 2 ? clampInt(65 + sources.filter((s) => s.tier === 'A' || s.tier === 'B').length * 10) : 55,
   411→  }
   412→}

   445→export async function ensureKnownNames() {
   446→  if (!knownNames) {
   447→    const rows = await dbp.aesthetic.findMany({ select: { id: true, name: true, aliases: true } })
   448→    knownNames = new Set()
   449→    knownIds = new Map()
   450→    for (const r of rows) {
   451→      const n = normalizeName(r.name)
   452→      knownNames.add(n)
   453→      if (!knownIds.has(n)) knownIds.set(n, r.id)
   454→      try {
   455→        const aliases = JSON.parse(r.aliases || '[]')
   456→        for (const a of aliases) {
   457→          if (typeof a === 'string') {
   458→            const an = normalizeName(a)
   459→            knownNames.add(an)
   460→            if (!knownIds.has(an)) knownIds.set(an, r.id)
   461→          }
   462→        }
   463→      } catch {}
   464→    }
   465→  }
   466→}

   468→export function isLikelyDupe(name: string, known: Set<string>): boolean {
   469→  const n = normalizeName(name)
   470→  if (!n) return true
   471→  if (known.has(n)) return true
   472→  // containment heuristic for near-duplicates ("vaporwave" vs "vaporwave art")
   473→  if (n.length >= 6) {
   474→    for (const k of known) {
   475→      if (k.length >= 6 && (k.includes(n) || n.includes(k))) return true
   476→    }
   477→  }
   478→  return false
   479→}

   481→export async function insertEntry(
   482→  entry: ValidEntry,
   483→  opts: { batchId?: string; status?: string; dataQuality?: string } = {}
   484→): Promise<{ ok: boolean; reason?: string; slug?: string }> {
   485→  if (!knownNames) await ensureKnownNames()
   486→  if (isLikelyDupe(entry.name, knownNames!)) return { ok: false, reason: 'duplicate' }
   487→  for (const a of entry.aliases) {
   488→    if (isLikelyDupe(a, knownNames!)) return { ok: false, reason: 'duplicate-alias' }
   489→  }
   490→
   491→  let slug = slugify(entry.name)
   492→  const existing = await dbp.aesthetic.findUnique({ where: { slug } })
   493→  if (existing) slug = `${slug}-${Date.now().toString(36).slice(-4)}`
   494→
   495→  const startYear = parseYear(entry.periodStart, 'start')
   496→  const endYear = parseYear(entry.periodEnd || entry.periodStart, 'end')
   497→
   498→  const sourceTiers = entry.sources.map((s) => s.tier)
   499→  const dataQuality =
   500→    opts.dataQuality ??
   501→    (sourceTiers.filter((t) => t === 'A' || t === 'B').length >= 2
   502→      ? 'well_documented'
   503→      : sourceTiers.some((t) => t === 'A' || t === 'B')
   504→        ? 'moderately_documented'
   505→        : sourceTiers.length > 0
   506→          ? 'emerging'
   507→          : 'interpretive')
   508→
   509→  try {
   510→    const created = await dbp.aesthetic.create({
   511→      data: {
   512→        slug,
   513→        name: entry.name,
   514→        aliases: JSON.stringify(entry.aliases),
   515→        category: entry.category,
   516→        subcategory: entry.subcategory,
   517→        establishment: entry.establishment,
   518→        status: opts.status ?? 'draft',
   519→        confidence: entry.confidence,
   520→        origin: entry.origin,
   521→        geography: entry.geography,
   522→        periodStart: entry.periodStart,
   523→        periodEnd: entry.periodEnd,
   524→        startYear,
   525→        endYear,
   526→        era: entry.era,
   527→        summary: entry.summary,
   528→        description: entry.description,
   529→        colors: JSON.stringify(entry.colors),
   530→        materials: JSON.stringify(entry.materials),
   531→        textures: JSON.stringify(entry.textures),
   532→        objects: JSON.stringify(entry.objects),
   533→        keyExamples: JSON.stringify(entry.keyExamples),
   534→        typography: JSON.stringify(entry.typography),
   535→        visualDNA: JSON.stringify(entry.visualDNA),
   536→        recipe: JSON.stringify(entry.recipe),
   537→        dnaAxes: JSON.stringify(entry.dnaAxes),
   538→        emotionProfile: JSON.stringify(entry.emotionProfile),
   539→        sources: JSON.stringify(entry.sources),
   540→        tags: JSON.stringify(entry.tags),
   541→        popularity: entry.popularity,
   542→        isNiche: entry.isNiche,
   543→        dataQuality,
   544→        batchId: opts.batchId ?? null,
   545→      },
   546→    })
   547→    knownNames!.add(normalizeName(entry.name))
   548→    knownIds!.set(normalizeName(entry.name), created.id)
   549→    for (const a of entry.aliases) {
   550→      const an = normalizeName(a)
   551→      knownNames!.add(an)
   552→      if (!knownIds!.has(an)) knownIds!.set(an, created.id)
   553→    }
   554→    await linkRelations(created.id, entry.name, entry.influences, entry.related)
   555→    return { ok: true, slug: created.slug }
   556→  } catch (e: any) {
   557→    return { ok: false, reason: `db: ${String(e.message).slice(0, 120)}` }
   558→  }
   559→}
   560→
   561→export async function linkRelations(
   562→  aestheticId: string,
   563→  aestheticName: string,
   564→  influences: string[],
   565→  related: string[]
   566→) {
   567→  if (!knownIds) await ensureKnownNames()
   568→  const targets: Array<{ type: string; names: string[] }> = [
   569→    { type: 'influenced_by', names: influences },
   570→    { type: 'related', names: related },
   571→  ]
   572→  for (const { type, names } of targets) {
   573→    for (const name of names) {
   574→      if (!name) continue
   575→      const n = normalizeName(name)
   576→      if (!n || n === normalizeName(aestheticName)) continue
   577→      const targetId = knownIds!.get(n)
   578→      if (targetId && targetId !== aestheticId) {
   579→        try {
   580→          await dbp.relation.create({ data: { fromId: aestheticId, toId: targetId, type } })
   581→        } catch {}
   582→      } else if (!targetId) {
   583→        try {
   584→          await dbp.backlogTerm.create({
   585→            data: { name: name.slice(0, 100), context: aestheticName, reason: `unresolved_${type}` },
   586→          })
   587→        } catch {}
   588→      }
   589→    }
   590→  }
   591→}
