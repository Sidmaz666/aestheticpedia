    23→function hexToRgb(hex: string): [number, number, number] | null {
    24→  let h = hex.replace('#', '').trim()
    25→  if (h.length === 3) h = h.split('').map((c) => c + c).join('')
    26→  if (!/^[0-9a-fA-F]{6}$/.test(h)) return null
    27→  return [parseInt(h.slice(0, 2), 16), parseInt(h.slice(2, 4), 16), parseInt(h.slice(4, 6), 16)]
    28→}
    29→
    30→function toVec(hex: string): [number, number, number] {
    31→  const rgb = hexToRgb(hex) ?? [137, 109, 59]
    32→  return [rgb[0] / 255, rgb[1] / 255, rgb[2] / 255]
    33→}
    34→
    35→function luminance(hex: string): number {
    36→  const rgb = hexToRgb(hex)
    37→  if (!rgb) return 0.5
    38→  const [r, g, b] = rgb.map((v) => {
    39→    const s = v / 255
    40→    return s <= 0.03928 ? s / 12.92 : Math.pow((s + 0.055) / 1.055, 2.4)
    41→  })
    42→  return 0.2126 * r + 0.7152 * g + 0.0722 * b
    43→}
    44→
    45→function mixHex(a: string, b: string, t: number): string {
    46→  const ca = hexToRgb(a)
    47→  const cb = hexToRgb(b)
    48→  if (!ca || !cb) return a
    49→  const c = ca.map((v, i) => Math.round(v + (cb[i] - v) * t))
    50→  return `#${c.map((v) => v.toString(16).padStart(2, '0')).join('')}`
    51→}
    52→
    53→function sortByLum(colors: ColorEntry[]): ColorEntry[] {
    54→  return [...colors].sort((a, b) => luminance(b.hex) - luminance(a.hex))
    55→}

    57→/** Deterministic per-entry seed. */
    58→function seedFrom(slug: string): number {
    59→  let h = 2166136261
    60→  for (let i = 0; i < slug.length; i++) {
    61→    h ^= slug.charCodeAt(i)
    62→    h = Math.imul(h, 16777619)
    63→  }
    64→  return (h >>> 0) % 9973
    65→}

   128→type GeoName = 'torus' | 'knot' | 'sphere'
   129→
   130→interface GeoData {
   131→  positions: Float32Array
   132→  normals: Float32Array
   133→  tangents: Float32Array
   134→  bitangents: Float32Array
   135→  uvs: Float32Array
   136→  indices: Uint16Array
   137→}
   138→
   139→function buildGeometry(geo: GeoName): GeoData {
   140→  const SEG = 110
   141→  const RINGS = 52
   142→  const positions: number[] = []
   143→  const normals: number[] = []
   144→  const tangents: number[] = []
   145→  const bitangents: number[] = []
   146→  const uvs: number[] = []
   147→  const indices: number[] = []
   148→  const R = 1.0
   149→  const r = geo === 'knot' ? 0.33 : 0.44
   150→  const p = geo === 'knot' ? 2 : 1
   151→  const q = geo === 'knot' ? 3 : 1
   152→  const cross = (a: [number, number, number], b: [number, number, number]): [number, number, number] => [
   153→    a[1] * b[2] - a[2] * b[1],
   154→    a[2] * b[0] - a[0] * b[2],
   155→    a[0] * b[1] - a[1] * b[0],
   156→  ]
   157→
   158→  // parametric centerline P(u, v), u,v in [0,1]
   159→  const P = (u: number, v: number): [number, number, number] => {
   160→    if (geo === 'sphere') {
   161→      const phi = v * Math.PI
   162→      const theta = u * Math.PI * 2
   163→      return [Math.sin(phi) * Math.cos(theta), Math.cos(phi), Math.sin(phi) * Math.sin(theta)]
   164→    }
   165→    const tu = u * Math.PI * 2 * p
   166→    if (geo === 'knot') {
   167→      const rad = R + 0.5 * Math.cos(q * tu / p)
   168→      return [rad * Math.cos(tu), 0.5 * Math.sin(q * tu / p), rad * Math.sin(tu)]
   169→    }
   170→    return [R * Math.cos(tu), 0, R * Math.sin(tu)]
   171→  }
   172→
   173→  const EPS = 1e-3
   174→  for (let i = 0; i <= SEG; i++) {
   175→    const u = i / SEG
   176→    for (let j = 0; j <= RINGS; j++) {
   177→      const v = j / RINGS
   178→      const c = P(u, v)
   179→      const pu = P(Math.min(1, u + EPS), v)
   180→      let T: [number, number, number] = [pu[0] - c[0], pu[1] - c[1], pu[2] - c[2]]
   181→      const tl = Math.hypot(...T) || 1
   182→      T = [T[0] / tl, T[1] / tl, T[2] / tl]
   183→      if (geo === 'sphere') {
   184→        // true sphere: point, outward normal, tangent frame from theta/phi
   185→        const n: [number, number, number] = [c[0], c[1], c[2]]
   186→        const Bn = cross(T, n)
   187→        positions.push(c[0], c[1], c[2])
   188→        normals.push(n[0], n[1], n[2])
   189→        tangents.push(T[0], T[1], T[2])
   190→        bitangents.push(Bn[0], Bn[1], Bn[2])
   191→        uvs.push(u * 3, v * 1.6)
   192→        continue
   193→      }
   194→      // tube: offset from centerline by r * (cos v * N0 + sin v * B0)
   195→      const tu = u * Math.PI * 2 * p
   196→      let N0: [number, number, number]
   197→      if (geo === 'knot') {
   198→        N0 = [-Math.sin(tu) * 0.5, Math.cos(q * tu / p) * 0.5, Math.cos(tu) * 0.5]
   199→      } else {
   200→        N0 = [0, 1, 0]
   201→      }
   202→      const n0l = Math.hypot(...N0) || 1
   203→      N0 = [N0[0] / n0l, N0[1] / n0l, N0[2] / n0l]
   204→      const B0: [number, number, number] = [
   205→        T[1] * N0[2] - T[2] * N0[1],
   206→        T[2] * N0[0] - T[0] * N0[2],
   207→        T[0] * N0[1] - T[1] * N0[0],
   208→      ]
   209→      const cosv = Math.cos(v * Math.PI * 2)
   210→      const sinv = Math.sin(v * Math.PI * 2)
   211→      const off: [number, number, number] = [
   212→        c[0] + (N0[0] * cosv + B0[0] * sinv) * r,
   213→        c[1] + (N0[1] * cosv + B0[1] * sinv) * r,
   214→        c[2] + (N0[2] * cosv + B0[2] * sinv) * r,
   215→      ]
   216→      const nrm: [number, number, number] = [N0[0] * cosv + B0[0] * sinv, N0[1] * cosv + B0[1] * sinv, N0[2] * cosv + B0[2] * sinv]
   217→      const nl = Math.hypot(...nrm) || 1
   218→      positions.push(...off)
   219→      normals.push(nrm[0] / nl, nrm[1] / nl, nrm[2] / nl)
   220→      tangents.push(...T)
   221→      bitangents.push(...B0)
   222→      uvs.push(u * 6, v * 2)
   223→    }
   224→  }
   225→  for (let i = 0; i < SEG; i++) {
   226→    for (let j = 0; j < RINGS; j++) {
   227→      const a = i * (RINGS + 1) + j
   228→      const b = a + RINGS + 1
   229→      indices.push(a, b, a + 1, b, b + 1, a + 1)
   230→    }
   231→  }
   232→  return {
   233→    positions: new Float32Array(positions),
   234→    normals: new Float32Array(normals),
   235→    tangents: new Float32Array(tangents),
   236→    bitangents: new Float32Array(bitangents),
   237→    uvs: new Float32Array(uvs),
   238→    indices: new Uint16Array(indices),
   239→  }
   240→}

   270→const FRAG_LIB = `
   271→precision highp float;
   272→varying vec3 vPos;
   273→varying vec3 vNormal;
   274→varying vec3 vTangent;
   275→varying vec3 vBitangent;
   276→varying vec2 vUv;
   277→
   278→uniform vec3 uColors[5];
   279→uniform float uSeed;
   280→uniform float uTime;
   281→uniform int uMaterial;
   282→
   283→vec2 uv0() { return vUv; }
   284→
   285→float hash(vec2 p) {
   286→  p = fract(p * vec2(234.34, 435.345));
   287→  p += dot(p, p + 34.23 + uSeed * 0.017);
   288→  return fract(p.x * p.y);
   289→}
   290→float noise(vec2 p) {
   291→  vec2 i = floor(p); vec2 f = fract(p);
   292→  vec2 u = f * f * (3.0 - 2.0 * f);
   293→  return mix(mix(hash(i), hash(i + vec2(1.0, 0.0)), u.x),
   294→             mix(hash(i + vec2(0.0, 1.0)), hash(i + vec2(1.0, 1.0)), u.x), u.y);
   295→}
   296→float fbm(vec2 p) {
   297→  float v = 0.0; float a = 0.5;
   298→  for (int i = 0; i < 5; i++) { v += a * noise(p); p *= 2.03; a *= 0.5; }
   299→  return v;
   300→}
   301→vec2 hash2(vec2 p) { return vec2(hash(p), hash(p + 19.19)); }
   302→
   303→// --- material height fields (uv space), also used for bump ---
   304→float matHeight(vec2 uv) {
   305→  if (uMaterial == 0) { // wood
   306→    vec2 q = uv * vec2(1.2, 14.0);
   307→    float w = fbm(q * 0.5) * 2.4;
   308→    float rings = sin((q.y * 0.55 + w) * 6.2831);
   309→    return rings * 0.5 + 0.5 + fbm(q * 3.0) * 0.12;
   310→  }
   311→  if (uMaterial == 1) { // marble veins
   312→    float v = fbm(uv * 2.6);
   313→    float veins = abs(sin((uv.x * 2.4 + uv.y * 1.4 + v * 4.6) * 3.1416));
   314→    return 1.0 - veins;
   315→  }
   316→  if (uMaterial == 2) { // brushed metal
   317→    return fbm(vec2(uv.x * 2.0, uv.y * 90.0)) * 0.7 + fbm(uv * 6.0) * 0.3;
   318→  }
   319→  if (uMaterial == 3) { // hammered dimples
   320→    vec2 g = uv * 14.0;
   321→    vec2 i = floor(g); vec2 f = fract(g) - 0.5;
   322→    float d = length(f);
   323→    return 1.0 - smoothstep(0.15, 0.55, d) + hash(i) * 0.1;
   324→  }
   325→  if (uMaterial == 4) { // rust blotches
   326→    return fbm(uv * 3.2 + uSeed) * 0.75 + fbm(vec2(uv.x * 6.0, uv.y * 0.7)) * 0.25;
   327→  }
   328→  if (uMaterial == 5) { // cloth weave
   329→    vec2 g = uv * vec2(120.0, 120.0);
   330→    return sin(g.x) * sin(g.y) * 0.5 + 0.5 + fbm(uv * 8.0) * 0.25;
   331→  }
   332→  if (uMaterial == 6) { return fbm(uv * 5.0); } // velvet pile
   333→  if (uMaterial == 7) { // ceramic crackle
   334→    vec2 g = uv * 26.0;
   335→    float cell = 1.0;
   336→    for (int y = -1; y <= 1; y++) for (int x = -1; x <= 1; x++) {
   337→      vec2 o = vec2(float(x), float(y));
   338→      vec2 c = hash2(floor(g) + o);
   339→      cell = min(cell, length(o + c - fract(g)));
   340→    }
   341→    return smoothstep(0.0, 0.22, cell);
   342→  }
   343→  if (uMaterial == 8) { // leather grain
   344→    return fbm(uv * 9.0) * 0.8 + noise(uv * 42.0) * 0.3;
   345→  }
   346→  if (uMaterial == 9) { return 0.5 + fbm(uv * 2.0) * 0.08; } // gloss plastic
   347→  if (uMaterial == 10) { // crt phosphor (animated)
   348→    float scan = sin(uv.y * 620.0) * 0.5 + 0.5;
   349→    float tri = sin(uv.x * 900.0) * 0.5 + 0.5;
   350→    float band = fract(uv.y * 1.6 + uTime * 0.09);
   351→    float roll = smoothstep(0.0, 0.18, band) * smoothstep(0.5, 0.2, band);
   352→    return scan * 0.4 + tri * 0.2 + roll * 0.7 + fbm(uv * 30.0 + uTime * 0.4) * 0.22;
   353→  }
   354→  if (uMaterial == 11) { // water ripples (animated)
   355→    return fbm(uv * 7.0 + vec2(uTime * 0.16, -uTime * 0.11)) + sin((uv.x + uv.y) * 26.0 + uTime * 1.4) * 0.06;
   356→  }
   357→  if (uMaterial == 12) { return fbm(uv * 3.0 + uTime * 0.05) + fbm(uv * 11.0) * 0.5; } // holo foil
   358→  if (uMaterial == 13) { // fur streaks
   359→    return fbm(vec2(uv.x * 70.0, uv.y * 8.0)) * 0.8 + noise(uv * 160.0) * 0.2;
   360→  }
   361→  if (uMaterial == 14) { // concrete pores
   362→    return fbm(uv * 5.0) * 0.7 + noise(uv * 60.0) * 0.3;
   363→  }
   364→  if (uMaterial == 15) { // terrazzo chips
   365→    vec2 g = uv * 16.0;
   366→    float d = 1.0;
   367→    for (int y = -1; y <= 1; y++) for (int x = -1; x <= 1; x++) {
   368→      vec2 o = vec2(float(x), float(y));
   369→      vec2 c = hash2(floor(g) + o);
   370→      d = min(d, length(o + c - fract(g)));
   371→    }
   372→    return smoothstep(0.0, 0.4, d) * 0.9 + fbm(uv * 40.0) * 0.1;
   373→  }
   374→  if (uMaterial == 16) { // ice cracks
   375→    vec2 g = uv * 12.0;
   376→    float cell = 1.0;
   377→    for (int y = -1; y <= 1; y++) for (int x = -1; x <= 1; x++) {
   378→      vec2 o = vec2(float(x), float(y));
   379→      vec2 c = hash2(floor(g) + o);
   380→      cell = min(cell, length(o + c - fract(g)));
   381→    }
   382→    return 1.0 - smoothstep(0.0, 0.14, cell) * 0.7;
   383→  }
   384→  if (uMaterial == 17) { return fbm(uv * 34.0) * 0.6 + noise(uv * 90.0) * 0.4; } // paper grain
   385→  if (uMaterial == 18) { return 0.5; } // glass — smooth, fresnel carries it
   386→  return fbm(uv * 3.0 + uTime * 0.07); // 19 neon fog
   387→}

   390→  if (uMaterial == 9 || uMaterial == 18) return 0.0;
   391→  if (uMaterial == 10) return 0.05;
   392→  if (uMaterial == 5) return 0.12;
   393→  if (uMaterial == 0) return 0.16;
   394→  return 0.22;

   397→// --- material base albedo & finish ---
   398→void matSurface(out vec3 albedo, out float spec, out float shin, out float metal, out float emissive) {
   399→  spec = 0.35; shin = 32.0; metal = 0.0; emissive = 0.0;
   400→  if (uMaterial == 0) {
   401→    float heart = smoothstep(0.25, 0.85, matHeight(uv0()));
   402→    albedo = mix(uColors[0], uColors[1], heart);
   403→    albedo = mix(albedo, uColors[2], fbm(uv0() * 22.0) * 0.22);
   404→    spec = 0.18; shin = 14.0;
   405→  } else if (uMaterial == 1) {
   406→    float vein = 1.0 - matHeight(uv0());
   407→    albedo = mix(uColors[0], uColors[1], fbm(uv0() * 1.6) * 0.5);
   408→    albedo = mix(albedo, uColors[2], smoothstep(0.55, 0.95, vein));
   409→    spec = 0.7; shin = 60.0;
   410→  } else if (uMaterial == 2) {
   411→    albedo = mix(uColors[0], uColors[1], matHeight(uv0()));
   412→    spec = 0.9; shin = 90.0; metal = 0.85;
   413→  } else if (uMaterial == 3) {
   414→    albedo = mix(uColors[0], uColors[1], matHeight(uv0()) * 0.7 + 0.15);
   415→    spec = 0.85; shin = 70.0; metal = 0.9;
   416→  } else if (uMaterial == 4) {
   417→    float h = matHeight(uv0());
   418→    albedo = mix(uColors[0], uColors[1], smoothstep(0.35, 0.75, h));
   419→    albedo = mix(albedo, uColors[2], smoothstep(0.62, 0.9, fbm(uv0() * 5.0)));
   420→    spec = 0.12; shin = 8.0;
   421→  } else if (uMaterial == 5) {
   422→    albedo = mix(uColors[0], uColors[1], matHeight(uv0()) * 0.85);
   423→    albedo = mix(albedo, uColors[2], fbm(uv0() * 3.0) * 0.3);
   424→    spec = 0.1; shin = 6.0;
   425→  } else if (uMaterial == 6) {
   426→    albedo = mix(uColors[0], uColors[1], matHeight(uv0()) * 0.6);
   427→    spec = 0.06; shin = 4.0;
   428→  } else if (uMaterial == 7) {
   429→    float crack = 1.0 - matHeight(uv0());
   430→    albedo = uColors[0];
   431→    albedo = mix(albedo, uColors[1], smoothstep(0.1, 0.6, fbm(uv0() * 2.4)));
   432→    albedo = mix(albedo, vec3(0.06, 0.05, 0.05), smoothstep(0.85, 1.0, crack) * 0.55);
   433→    spec = 0.6; shin = 80.0;
   434→  } else if (uMaterial == 8) {
   435→    albedo = mix(uColors[0], uColors[1], matHeight(uv0()) * 0.8);
   436→    spec = 0.28; shin = 18.0;
   437→  } else if (uMaterial == 9) {
   438→    albedo = mix(uColors[0], uColors[1], 0.35);
   439→    albedo = mix(albedo, uColors[2], fbm(uv0() * 2.0) * 0.2);
   440→    spec = 0.95; shin = 110.0;
   441→  } else if (uMaterial == 10) {
   442→    float glow = matHeight(uv0());
   443→    albedo = mix(uColors[0] * 0.12, uColors[1], glow);
   444→    emissive = glow * 0.85;
   445→    spec = 0.7; shin = 90.0;
   446→  } else if (uMaterial == 11) {
   447→    albedo = mix(uColors[0], uColors[1], 0.55);
   448→    spec = 1.1; shin = 140.0; metal = 0.25;
   449→  } else if (uMaterial == 12) {
   450→    albedo = uColors[0];
   451→    spec = 1.0; shin = 120.0; metal = 0.6;
   452→  } else if (uMaterial == 13) {
   453→    albedo = mix(uColors[0], uColors[1], matHeight(uv0()));
   454→    spec = 0.22; shin = 10.0;
   455→  } else if (uMaterial == 14) {
   456→    albedo = mix(uColors[0], uColors[1], matHeight(uv0()) * 0.7);
   457→    albedo = mix(albedo, uColors[2], 0.25);
   458→    spec = 0.08; shin = 6.0;
   459→  } else if (uMaterial == 15) {
   460→    vec2 g = uv0() * 16.0;
   461→    vec3 chip = uColors[1];
   462→    float d = 1.0;
   463→    vec2 base = floor(g);
   464→    for (int y = -1; y <= 1; y++) for (int x = -1; x <= 1; x++) {
   465→      vec2 o = vec2(float(x), float(y));
   466→      vec2 cc = base + o + hash2(base + o);
   467→      float dd = length(cc - g);
   468→      if (dd < d) { d = dd; chip = mix(uColors[1], uColors[2], hash(cc)); }
   469→    }
   470→    albedo = mix(chip, uColors[0], smoothstep(0.0, 0.4, d));
   471→    spec = 0.5; shin = 46.0;
   472→  } else if (uMaterial == 16) {
   473→    float crack = 1.0 - matHeight(uv0());
   474→    albedo = mix(uColors[0], uColors[1], 0.5);
   475→    albedo = mix(albedo, uColors[2], smoothstep(0.7, 1.0, crack));
   476→    spec = 0.9; shin = 100.0;
   477→  } else if (uMaterial == 17) {
   478→    albedo = uColors[0];
   479→    albedo = mix(albedo, uColors[1], matHeight(uv0()) * 0.22);
   480→    spec = 0.04; shin = 4.0;
   481→  } else if (uMaterial == 18) {
   482→    albedo = mix(uColors[0], uColors[1], 0.4);
   483→    spec = 1.2; shin = 160.0;
   484→  } else {
   485→    float glow = matHeight(uv0());
   486→    albedo = mix(uColors[0], uColors[1], glow);
   487→    emissive = pow(glow, 1.6) * 1.1;
   488→    spec = 0.4; shin = 40.0;
   489→  }
   490→}
   491→`

   493→const FRAG_HELPERS = `
   494→vec3 cameraPos() { return vec3(0.0, 0.0, 3.4); }
   495→vec3 lightDir1() { return vec3(0.55, 0.7, 0.6); }
   496→vec3 lightTint() { return vec3(1.0, 0.98, 0.94); }
   497→vec3 fresnelTint() { return mix(uColors[1], vec3(1.0), 0.4); }
   498→`
   499→
   500→const FRAG_MAIN = `
   501→void main() {
   502→  vec3 N = normalize(vNormal);
   503→  vec3 V = normalize(cameraPos() - vPos);
   504→
   505→  // bump from height field along the surface tangent frame
   506→  float eps = 0.004;
   507→  float h  = matHeight(vUv);
   508→  float hx = matHeight(vUv + vec2(eps, 0.0));
   509→  float hy = matHeight(vUv + vec2(0.0, eps));
   510→  float bumpAmt = bumpScaleFor() * 46.0;
   511→  N = normalize(N - (vTangent * (hx - h) + vBitangent * (hy - h)) * bumpAmt);
   512→
   513→  vec3 albedo; float spec; float shin; float metal; float emissive;
   514→  matSurface(albedo, spec, shin, metal, emissive);
   515→
   516→  if (uMaterial == 12) { // thin-film iridescence
   517→    float f = pow(1.0 - abs(dot(N, V)), 2.2);
   518→    float band = sin((vUv.x + vUv.y) * 12.0 + f * 9.0 + uTime * 0.35);
   519→    albedo = mix(mix(uColors[1], uColors[2], 0.5 + 0.5 * band), uColors[3], f * 0.8);
   520→    albedo = mix(albedo, uColors[4], 0.35 * (0.5 + 0.5 * sin(f * 12.0)));
   521→  }
   522→
   523→  vec3 L1 = normalize(lightDir1());
   524→  vec3 L2 = normalize(vec3(-0.6, -0.2, 0.5));
   525→  float dif = max(dot(N, L1), 0.0);
   526→  float dif2 = max(dot(N, L2), 0.0) * 0.28;
   527→  vec3 H1 = normalize(L1 + V);
   528→  float sp = pow(max(dot(N, H1), 0.0), shin) * spec * (1.0 - metal * 0.25);
   529→  float fres = pow(1.0 - abs(dot(N, V)), 3.2);
   530→  vec3 col = albedo * (0.34 + dif * 0.95 + dif2) + vec3(sp) * lightTint();
   531→  if (uMaterial == 6) col += albedo * pow(1.0 - abs(dot(N, V)), 1.6) * 0.55; // velvet sheen
   532→  col += fres * fresnelTint() * (0.28 + metal * 0.6);
   533→  col += albedo * emissive;
   534→  col = pow(clamp(col, 0.0, 1.4), vec3(0.92));
   535→  gl_FragColor = vec4(col, 1.0);
   536→}
   537→`

   554→  const [supported, setSupported] = useState(true)
   555→  const reducedMotion = useMemo(
   556→    () => typeof window !== 'undefined' && window.matchMedia('(prefers-reduced-motion: reduce)').matches,
   557→    []
   558→  )
   559→
   560→  const palette = useMemo(() => {
   561→    const valid = sortByLum(a.colors.filter((c: ColorEntry) => hexToRgb(c.hex)))
   562→    const light = valid[0]?.hex ?? '#efe9dd'
   563→    const dark = valid[valid.length - 1]?.hex ?? '#2f2a26'

   565→    if (valid.length === 4) return [...valid.map((c: ColorEntry) => c.hex), mixHex(light, '#ffffff', 0.35)]

   567→    if (valid.length === 2) return [...valid.map((c: ColorEntry) => c.hex), mixHex(light, dark, 0.5), mixHex(light, '#ffffff', 0.25), mixHex(dark, '#000000', 0.25)]
   568→    if (valid.length === 1) return [valid[0].hex, mixHex(valid[0].hex, '#ffffff', 0.4), mixHex(valid[0].hex, '#000000', 0.35), mixHex(valid[0].hex, '#8a6d3b', 0.3), mixHex(valid[0].hex, '#ffffff', 0.7)]
   569→    return [light, '#8a6d3b', '#5b5147', '#c9b99a', dark]
   570→  }, [a.colors])
   571→
   572→  const stateRef = useRef<{
   573→    rotX: number
   574→    rotY: number
   575→    zoom: number
   576→    dragging: boolean
   577→    lastX: number
   578→    lastY: number

   580→    raf: number
   581→    start: number

   585→  }>({ rotX: -0.35, rotY: 0.6, zoom: 1, dragging: false, lastX: 0, lastY: 0, autoSpin: true, raf: 0, start: 0, curMaterial: 0, curGeo: 'knot', draw: null })
   586→
   587→  useEffect(() => {
   588→    const canvas = canvasRef.current
   589→    if (!canvas) return
   590→    const gl = canvas.getContext('webgl', { antialias: true, alpha: false })
   591→    if (!gl) { setSupported(false); return }
   592→
   593→    const compile = (type: number, src: string) => {
   594→      const sh = gl.createShader(type)!
   595→      gl.shaderSource(sh, src)
   596→      gl.compileShader(sh)
   597→      if (!gl.getShaderParameter(sh, gl.COMPILE_STATUS)) {
   598→        console.error('shader compile error:', gl.getShaderInfoLog(sh))
   599→        return null
   600→      }
   601→      return sh
   602→    }
   603→    const vs = compile(gl.VERTEX_SHADER, VERT)
   604→    const fs = compile(gl.FRAGMENT_SHADER, FRAG_FULL)
   605→    if (!vs || !fs) { setSupported(false); return }
   606→    const program = gl.createProgram()!
   607→    gl.attachShader(program, vs)
   608→    gl.attachShader(program, fs)
   609→    gl.linkProgram(program)
   610→    if (!gl.getProgramParameter(program, gl.LINK_STATUS)) {
   611→      console.error('program link error:', gl.getProgramInfoLog(program))
   612→      setSupported(false)
   613→      return
   614→    }
   615→    gl.useProgram(program)
   616→
   617→    const loc = (name: string) => gl.getAttribLocation(program, name)
   618→    const uniforms = {
   619→      uProj: gl.getUniformLocation(program, 'uProj'),
   620→      uView: gl.getUniformLocation(program, 'uView'),
   621→      uModel: gl.getUniformLocation(program, 'uModel'),
   622→      uNormalMat: gl.getUniformLocation(program, 'uNormalMat'),
   623→      uColors: gl.getUniformLocation(program, 'uColors'),
   624→      uSeed: gl.getUniformLocation(program, 'uSeed'),
   625→      uTime: gl.getUniformLocation(program, 'uTime'),
   626→      uMaterial: gl.getUniformLocation(program, 'uMaterial'),
   627→    }
   628→
   629→    const counts = {} as Record<GeoName, number>
   630→    const drawSetup: Record<GeoName, () => void> = {} as any
   631→    const makeGeo = (name: GeoName) => {
   632→      const g = buildGeometry(name)
   633→      const bind = (data: Float32Array, attr: string, size: number) => {
   634→        const buf = gl.createBuffer()!
   635→        gl.bindBuffer(gl.ARRAY_BUFFER, buf)
   636→        gl.bufferData(gl.ARRAY_BUFFER, data, gl.STATIC_DRAW)
   637→        const l = loc(attr)
   638→        gl.enableVertexAttribArray(l)
   639→        gl.vertexAttribPointer(l, size, gl.FLOAT, false, 0, 0)
   640→      }
   641→      const posBuf = gl.createBuffer()!
   642→      gl.bindBuffer(gl.ARRAY_BUFFER, posBuf)
   643→      gl.bufferData(gl.ARRAY_BUFFER, g.positions, gl.STATIC_DRAW)
   644→      const nrmBuf = gl.createBuffer()!
   645→      gl.bindBuffer(gl.ARRAY_BUFFER, nrmBuf)
   646→      gl.bufferData(gl.ARRAY_BUFFER, g.normals, gl.STATIC_DRAW)
   647→      const tanBuf = gl.createBuffer()!
   648→      gl.bindBuffer(gl.ARRAY_BUFFER, tanBuf)
   649→      gl.bufferData(gl.ARRAY_BUFFER, g.tangents, gl.STATIC_DRAW)
   650→      const bitBuf = gl.createBuffer()!
   651→      gl.bindBuffer(gl.ARRAY_BUFFER, bitBuf)
   652→      gl.bufferData(gl.ARRAY_BUFFER, g.bitangents, gl.STATIC_DRAW)
   653→      const uvBuf = gl.createBuffer()!
   654→      gl.bindBuffer(gl.ARRAY_BUFFER, uvBuf)
   655→      gl.bufferData(gl.ARRAY_BUFFER, g.uvs, gl.STATIC_DRAW)
   656→      const idxBuf = gl.createBuffer()!
   657→      gl.bindBuffer(gl.ELEMENT_ARRAY_BUFFER, idxBuf)
   658→      gl.bufferData(gl.ELEMENT_ARRAY_BUFFER, g.indices, gl.STATIC_DRAW)
   659→      counts[name] = g.indices.length
   660→      const lp = loc('aPos'); const ln = loc('aNormal'); const lt = loc('aTangent'); const lb = loc('aBitangent'); const lu = loc('aUv')
   661→      drawSetup[name] = () => {
   662→        gl.bindBuffer(gl.ARRAY_BUFFER, posBuf)
   663→        gl.enableVertexAttribArray(lp)
   664→        gl.vertexAttribPointer(lp, 3, gl.FLOAT, false, 0, 0)
   665→        gl.bindBuffer(gl.ARRAY_BUFFER, nrmBuf)
   666→        gl.enableVertexAttribArray(ln)
   667→        gl.vertexAttribPointer(ln, 3, gl.FLOAT, false, 0, 0)
   668→        gl.bindBuffer(gl.ARRAY_BUFFER, tanBuf)
   669→        gl.enableVertexAttribArray(lt)
   670→        gl.vertexAttribPointer(lt, 3, gl.FLOAT, false, 0, 0)
   671→        gl.bindBuffer(gl.ARRAY_BUFFER, bitBuf)
   672→        gl.enableVertexAttribArray(lb)
   673→        gl.vertexAttribPointer(lb, 3, gl.FLOAT, false, 0, 0)
   674→        gl.bindBuffer(gl.ARRAY_BUFFER, uvBuf)
   675→        gl.enableVertexAttribArray(lu)
   676→        gl.vertexAttribPointer(lu, 2, gl.FLOAT, false, 0, 0)
   677→        gl.bindBuffer(gl.ELEMENT_ARRAY_BUFFER, idxBuf)
   678→      }
   679→    }
   680→    ;(['knot', 'torus', 'sphere'] as GeoName[]).forEach(makeGeo)
   681→
   682→    const st = stateRef.current
   683→    st.autoSpin = !reducedMotion
   684→    st.start = performance.now()
   685→    st.curMaterial = Math.max(0, MATERIAL_ORDER.indexOf(material))
   686→    st.curGeo = geo
   687→    st.draw = (name: GeoName) => drawSetup[name]()
   688→
   689→    const resize = () => {
   690→      const dpr = Math.min(window.devicePixelRatio || 1, 2)
   691→      const rect = canvas.getBoundingClientRect()
   692→      canvas.width = Math.max(2, Math.round(rect.width * dpr))
   693→      canvas.height = Math.max(2, Math.round(rect.height * dpr))
   694→      gl.viewport(0, 0, canvas.width, canvas.height)
   695→    }
   696→    resize()
   697→    const ro = new ResizeObserver(resize)
   698→    ro.observe(canvas)
   699→
   700→    const mat4Perspective = (fovy: number, aspect: number, near: number, far: number) => {
   701→      const f = 1 / Math.tan(fovy / 2)
   702→      const nf = 1 / (near - far)
   703→      return new Float32Array([f / aspect, 0, 0, 0, 0, f, 0, 0, 0, 0, (far + near) * nf, -1, 0, 0, 2 * far * near * nf, 0])
   704→    }
   705→    const mat4LookAt = (eye: number[], center: number[], up: number[]) => {
   706→      const z = eye.map((e, i) => e - center[i])
   707→      const zl = Math.hypot(...z); const Z = z.map((v) => v / zl)
   708→      const x = [up[1] * Z[2] - up[2] * Z[1], up[2] * Z[0] - up[0] * Z[2], up[0] * Z[1] - up[1] * Z[0]]
   709→      const xl = Math.hypot(...x); const X = x.map((v) => v / xl)
   710→      const Y = [Z[1] * X[2] - Z[2] * X[1], Z[2] * X[0] - Z[0] * X[2], Z[0] * X[1] - Z[1] * X[0]]
   711→      return new Float32Array([X[0], Y[0], Z[0], 0, X[1], Y[1], Z[1], 0, X[2], Y[2], Z[2], 0,
   712→        -(X[0] * eye[0] + X[1] * eye[1] + X[2] * eye[2]),
   713→        -(Y[0] * eye[0] + Y[1] * eye[1] + Y[2] * eye[2]),
   714→        -(Z[0] * eye[0] + Z[1] * eye[1] + Z[2] * eye[2]), 1])
   715→    }
   716→    const rotXYZ = (rx: number, ry: number) => {
   717→      const cx = Math.cos(rx), sx = Math.sin(rx), cy = Math.cos(ry), sy = Math.sin(ry)
   718→      return new Float32Array([cy, 0, -sy, 0, sy * sx, cx, cy * sx, 0, sy * cx, -sx, cy * cx, 0, 0, 0, 0, 1])
   719→    }
   720→    const normalFromModel = (m: Float32Array) => new Float32Array([m[0], m[1], m[2], m[4], m[5], m[6], m[8], m[9], m[10]])
   721→
   722→    const flat: number[] = []
   723→    palette.forEach((hex) => flat.push(...toVec(hex)))
   724→    while (flat.length < 15) flat.push(0.5, 0.5, 0.5)
   725→

   727→      const t = (performance.now() - st.start) / 1000
   728→      if (st.autoSpin && !st.dragging) st.rotY += 0.0035

   730→      gl.clear(gl.COLOR_BUFFER_BIT | gl.DEPTH_BUFFER_BIT)
   731→      gl.enable(gl.DEPTH_TEST)
   732→
   733→      const aspect = canvas.width / Math.max(1, canvas.height)
   734→      const proj = mat4Perspective(Math.PI / 4, aspect, 0.1, 100)
   735→      const view = mat4LookAt([0, 0, 3.4 / st.zoom], [0, 0, 0], [0, 1, 0])
   736→      const model = rotXYZ(st.rotX, st.rotY)
   737→      st.draw!(st.curGeo)
   738→
   739→      gl.uniformMatrix4fv(uniforms.uProj, false, proj)
   740→      gl.uniformMatrix4fv(uniforms.uView, false, view)
   741→      gl.uniformMatrix4fv(uniforms.uModel, false, model)
   742→      gl.uniformMatrix3fv(uniforms.uNormalMat, false, normalFromModel(model))
   743→      gl.uniform3fv(uniforms.uColors, new Float32Array(flat))
   744→      gl.uniform1f(uniforms.uSeed, seedFrom(a.slug))
   745→      gl.uniform1f(uniforms.uTime, reducedMotion ? 2.5 : t)
   746→      gl.uniform1i(uniforms.uMaterial, st.curMaterial)
   747→      gl.drawElements(gl.TRIANGLES, counts[st.curGeo], gl.UNSIGNED_SHORT, 0)
   748→
   749→      st.raf = requestAnimationFrame(render)
   750→    }
   751→    st.raf = requestAnimationFrame(render)
   752→
   753→    const down = (e: PointerEvent) => {
   754→      st.dragging = true
   755→      st.lastX = e.clientX; st.lastY = e.clientY
   756→      canvas.setPointerCapture(e.pointerId)
   757→    }
   758→    const move = (e: PointerEvent) => {
   759→      if (!st.dragging) return
   760→      st.rotY += (e.clientX - st.lastX) * 0.008
   761→      st.rotX += (e.clientY - st.lastY) * 0.008
   762→      st.rotX = Math.max(-1.4, Math.min(1.4, st.rotX))
   763→      st.lastX = e.clientX; st.lastY = e.clientY
   764→    }
   765→    const up = () => { st.dragging = false }
   766→    const wheel = (e: WheelEvent) => {
   767→      e.preventDefault()
   768→      st.zoom = Math.max(0.6, Math.min(2.6, st.zoom * (e.deltaY > 0 ? 0.92 : 1.08)))
   769→    }
   770→    canvas.addEventListener('pointerdown', down)
   771→    canvas.addEventListener('pointermove', move)
   772→    canvas.addEventListener('pointerup', up)
   773→    canvas.addEventListener('pointercancel', up)
   774→    canvas.addEventListener('wheel', wheel, { passive: false })
   775→
   776→    return () => {
   777→      cancelAnimationFrame(st.raf)
   778→      ro.disconnect()
   779→      canvas.removeEventListener('pointerdown', down)
   780→      canvas.removeEventListener('pointermove', move)
   781→      canvas.removeEventListener('pointerup', up)
   782→      canvas.removeEventListener('pointercancel', up)
   783→      canvas.removeEventListener('wheel', wheel)
   784→    }
   785→  }, [a.slug, reducedMotion])
   786→
   787→  useEffect(() => {
   788→    const st = stateRef.current
   789→    if (!st) return
   790→    st.curMaterial = Math.max(0, MATERIAL_ORDER.indexOf(material))
   791→    if (expanded) st.zoom = Math.max(st.zoom, 1.3)
   792→  }, [material, expanded])
   793→
   794→  useEffect(() => {
   795→    const st = stateRef.current
   796→    if (st) st.curGeo = geo
   797→  }, [geo])
   798→
   799→  if (!supported) {
   800→    return (
   801→      <div className={`flex h-48 items-center justify-center rounded-lg border border-stone-200 bg-stone-100 px-6 text-center text-xs text-stone-500 ${className ?? ''}`} role="img" aria-label="3D material preview unavailable">
   802→        3D shader preview needs WebGL. The material vocabulary for this entry is still documented below and in the texture swatches.
   803→      </div>
   804→    )
   805→  }
   806→
   807→  return (
