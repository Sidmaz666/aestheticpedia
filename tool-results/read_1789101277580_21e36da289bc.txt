   615→   615→  note?: string

   620→   620→function RelationsSection({
   621→   621→  relations,
   622→   622→  onNavigate,
   623→   623→}: {
   624→   624→  relations: { outgoing: RelationEdge[]; incoming: RelationEdge[] }
   625→   625→  onNavigate: (slug: string) => void
   626→   626→}) {
   627→   627→  if (relations.outgoing.length === 0 && relations.incoming.length === 0) {
   628→   628→    return (
   629→   629→      <section aria-label="Relationships">
   630→   630→        <SectionHeading>Relationships</SectionHeading>
   631→   631→        <p className="text-sm text-stone-500">No documented relationships yet — the pipeline may link this entry later.</p>
   632→   632→      </section>
   633→   633→    )
   634→   634→  }
   635→   635→
   636→   636→  const group = (edges: RelationEdge[]) => {
   637→   637→    const map = new Map<string, RelationEdge[]>()
   638→   638→    for (const e of edges) {
   639→   639→      const list = map.get(e.type) ?? []
   640→   640→      list.push(e)
   641→   641→      map.set(e.type, list)
   642→   642→    }
   643→   643→    return [...map.entries()]
   644→   644→  }
   645→   645→
   646→   646→  return (
   647→   647→    <section aria-label="Relationships">
   648→   648→      <SectionHeading hint="The aesthetic family tree.">Relationships</SectionHeading>
   649→   649→      <div className="space-y-4">
   650→   650→        {relations.outgoing.length > 0 && (
   651→   651→          <div>
   652→   652→            <p className="mb-1.5 flex items-center gap-1.5 text-[11px] font-medium uppercase tracking-[0.14em] text-stone-500">
   653→   653→              <ArrowRight className="h-3 w-3" aria-hidden="true" /> Outgoing
   654→   654→            </p>
   655→   655→            <div className="space-y-2.5">
   656→   656→              {group(relations.outgoing).map(([type, edges]) => (
   657→   657→                <div key={type} className="flex flex-wrap items-center gap-1.5">
   658→   658→                  <span className="mr-1 min-w-fit text-xs capitalize text-stone-400">
   659→   659→                    {labelize(type)}:
   660→   660→                  </span>
   661→   661→                  {edges.map((e) =>
   662→   662→                    e.target ? (
   663→   663→                      <RelationChip
   664→   664→                        key={`${e.target.slug}-${type}`}
   665→   665→                        name={e.target.name}
   666→   666→                        note={e.note}
   667→   667→                        onClick={() => onNavigate(e.target!.slug)}
   668→   668→                      />
   669→   669→                    ) : null
   670→   670→                  )}
   671→   671→                </div>
   672→   672→              ))}
   673→   673→            </div>
   674→   674→          </div>
   675→   675→        )}
   676→   676→        {relations.incoming.length > 0 && (
   677→   677→          <div>
   678→   678→            <p className="mb-1.5 flex items-center gap-1.5 text-[11px] font-medium uppercase tracking-[0.14em] text-stone-500">
   679→   679→              <ArrowLeft className="h-3 w-3" aria-hidden="true" /> Incoming
   680→   680→            </p>
   681→   681→            <div className="space-y-2.5">
   682→   682→              {group(relations.incoming).map(([type, edges]) => (
   683→   683→                <div key={type} className="flex flex-wrap items-center gap-1.5">
   684→   684→                  <span className="mr-1 min-w-fit text-xs capitalize text-stone-400">
   685→   685→                    {labelize(type)}:
   686→   686→                  </span>
   687→   687→                  {edges.map((e) =>
   688→   688→                    e.source ? (
   689→   689→                      <RelationChip
   690→   690→                        key={`${e.source.slug}-${type}`}
   691→   691→                        name={e.source.name}
   692→   692→                        note={e.note}
   693→   693→                        onClick={() => onNavigate(e.source!.slug)}
   694→   694→                      />
   695→   695→                    ) : null
   696→   696→                  )}
   697→   697→                </div>
   698→   698→              ))}
   699→   699→            </div>
   700→   700→          </div>
   701→   701→        )}
   702→   702→      </div>
   703→   703→    </section>
   704→   704→  )
