    25→function hexToRgb(hex: string): [number, number, number] | null {
    26→  let h = hex.replace('#', '').trim()
    27→  if (h.length === 3) h = h.split('').map((c) => c + c).join('')
    28→  if (!/^[0-9a-fA-F]{6}$/.test(h)) return null
    29→  return [parseInt(h.slice(0, 2), 16), parseInt(h.slice(2, 4), 16), parseInt(h.slice(4, 6), 16)]
    30→}
    31→
    32→/** Relative luminance 0 (dark) – 1 (light). */
    33→function luminance(hex: string): number {
    34→  const rgb = hexToRgb(hex)
    35→  if (!rgb) return 0.5
    36→  const [r, g, b] = rgb.map((v) => {
    37→    const s = v / 255
    38→    return s <= 0.03928 ? s / 12.92 : Math.pow((s + 0.055) / 1.055, 2.4)
    39→  })
    40→  return 0.2126 * r + 0.7152 * g + 0.0722 * b
    41→}
    42→
    43→function saturation(hex: string): number {
    44→  const rgb = hexToRgb(hex)
    45→  if (!rgb) return 0
    46→  const [r, g, b] = rgb.map((v) => v / 255)
    47→  const max = Math.max(r, g, b)
    48→  const min = Math.min(r, g, b)
    49→  if (max === min) return 0
    50→  const l = (max + min) / 2
    51→  return l > 0.5 ? (max - min) / (2 - max - min) : (max - min) / (max + min)
    52→}
    53→
    54→function rgba(hex: string, alpha: number): string {
    55→  const rgb = hexToRgb(hex) ?? [137, 109, 59]
    56→  return `rgba(${rgb[0]}, ${rgb[1]}, ${rgb[2]}, ${alpha})`
    57→}
    58→
    59→/** Black or white text color for a given background. */
    60→function readableOn(hex: string): string {
    61→  return luminance(hex) > 0.45 ? '#1c1917' : '#fdfcf8'
    62→}

    84→/** Derive demo roles (bg/surface/ink/accent) from the entry palette. */
    85→function derivePalette(colors: ColorEntry[]): DemoPalette {
    86→  const valid = colors.filter((c) => hexToRgb(c.hex))
    87→  if (valid.length < 2) return FALLBACK
    88→  const sorted = [...valid].sort((a, b) => luminance(b.hex) - luminance(a.hex))
    89→  const bg = sorted[0].hex
    90→  const ink = sorted[sorted.length - 1].hex
    91→  // accent: most saturated color that is neither bg nor ink
    92→  const mid = valid.filter((c) => c.hex !== bg && c.hex !== ink)
    93→  const accent =
    94→    mid.length > 0
    95→      ? [...mid].sort((a, b) => saturation(b.hex) * (1 - Math.abs(luminance(b.hex) - 0.5)) - saturation(a.hex) * (1 - Math.abs(luminance(a.hex) - 0.5)))[0].hex
    96→      : sorted[Math.floor(sorted.length / 2)].hex
    97→  const surface = sorted.length > 2 ? sorted[1].hex : mix(bg, ink, 0.08)
    98→  const support = mid.find((c) => c.hex !== accent)?.hex ?? mix(accent, bg, 0.3)
    99→  const warm = isWarm(accent)
   100→  return { bg, surface, ink, accent, support, warm, light: luminance(bg) > 0.5 }
   101→}
   102→
   103→function mix(a: string, b: string, t: number): string {
   104→  const ca = hexToRgb(a)
   105→  const cb = hexToRgb(b)
   106→  if (!ca || !cb) return a
   107→  const c = ca.map((v, i) => Math.round(v + (cb[i] - v) * t))
   108→  return `#${c.map((v) => v.toString(16).padStart(2, '0')).join('')}`
   109→}
   110→
   111→function isWarm(hex: string): boolean {
   112→  const rgb = hexToRgb(hex)
   113→  if (!rgb) return true
   114→  const [r, g, b] = rgb
   115→  return r + r * 0.2 + g * 0.3 >= b
   116→}

   120→const SERIF = 'ui-serif, Georgia, Cambria, "Times New Roman", serif'
   121→const SANS = 'ui-sans-serif, system-ui, "Helvetica Neue", Arial, sans-serif'
   122→const MONO = 'ui-monospace, "SF Mono", Menlo, Consolas, monospace'

   125→function fontFor(text: string, fallback: 'serif' | 'sans'): string {
   126→  const t = (text || '').toLowerCase()
   127→  if (/mono|typewriter|courier|terminal/.test(t)) return MONO
   128→  if (/script|cursive|hand|brush letter|calligrap/.test(t)) return SCRIPT
   129→  if (/serif|didone|garamond|roman|slab|bodoni|clarendon|garamond/.test(t) && !/sans/.test(t)) return SERIF
   130→  if (/sans|grotesque|geometric|gothic|helvetica|futura|swiss|arial|neo-grotesque|rounded/.test(t)) return SANS
   131→  if (fallback === 'serif') return SERIF
   132→  return SANS
   133→}

   142→function displayFont(a: AestheticFull): FontSpec {
   143→  const t = `${a.typography.display ?? ''} ${a.typography.notes ?? ''} ${a.name}`.toLowerCase()
   144→  const stack = fontFor(t, ['Internet Aesthetic', 'Web & UI Design', 'Game & Pixel Aesthetic', 'Graphic Design', 'Technology & Retrofuturism', 'Music & Sonic Culture'].includes(a.category) ? 'sans' : 'serif')
   145→  const deco = /art deco|deco|geometric|broadway|display/.test(t)
   146→  const mono = stack === MONO
   147→  return {
   148→    stack,
   149→    tracking: deco || mono ? '0.12em' : /serif/.test(stack) ? '0.01em' : '-0.01em',
   150→    transform: deco || /all caps|uppercase|poster/.test(t) ? 'uppercase' : 'none',
   151→    weight: mono ? 500 : 700,
   152→  }
   153→}
   154→
   155→function bodyFont(a: AestheticFull): string {
   156→  return fontFor(`${a.typography.body ?? ''}`, 'sans')
   157→}

   166→/** Deterministic keyword-driven CSS texture tinted by the palette. */
   167→function texturePattern(texture: string, p: DemoPalette): TexturePattern {
   168→  const t = texture.toLowerCase()
   169→  const inkA = rgba(p.ink, 0.1)
   170→  const inkSoft = rgba(p.ink, 0.05)
   171→  const accentA = rgba(p.accent, 0.12)
   172→  if (/wood|timber|plank|oak|walnut|bamboo|driftwood/.test(t)) {
   173→    return {
   174→      backgroundImage: `repeating-linear-gradient(92deg, ${inkA} 0 1px, transparent 1px 6px), repeating-linear-gradient(90deg, ${inkSoft} 0 2px, transparent 2px 13px), linear-gradient(180deg, ${rgba(p.support, 0.18)}, ${rgba(p.accent, 0.08)})`,
   175→      backgroundSize: 'auto, auto, auto',
   176→    }
   177→  }
   178→  if (/linen|weave|woven|fabric|textile|canvas|burlap|tweed|boucl|herringbone|ikat/.test(t)) {
   179→    return {
   180→      backgroundImage: `repeating-linear-gradient(0deg, ${inkSoft} 0 1px, transparent 1px 4px), repeating-linear-gradient(90deg, ${inkSoft} 0 1px, transparent 1px 4px)`,
   181→      backgroundSize: 'auto',
   182→    }
   183→  }
   184→  if (/velvet|plush|fleece|suede|chenille|fur/.test(t)) {
   185→    return {
   186→      backgroundImage: `radial-gradient(ellipse at 30% 20%, ${rgba(p.accent, 0.22)}, transparent 60%), radial-gradient(ellipse at 70% 80%, ${rgba(p.ink, 0.1)}, transparent 55%)`,
   187→      backgroundSize: 'auto',
   188→    }
   189→  }
   190→  if (/concrete|stone|plaster|cement|stucco|terrazzo|marble|granite|mineral/.test(t)) {
   191→    return {
   192→      backgroundImage: `radial-gradient(${inkA} 0.8px, transparent 0.9px), radial-gradient(${inkSoft} 0.7px, transparent 0.8px), linear-gradient(160deg, ${rgba(p.support, 0.12)}, transparent)`,
   193→      backgroundSize: '7px 7px, 11px 11px, auto',
   194→    }
   195→  }
   196→  if (/metal|steel|chrome|brushed|alumin|foil|tin|silver|gold|gilt|brass|copper|bronze|patina/.test(t)) {
   197→    return {
   198→      backgroundImage: `linear-gradient(105deg, transparent 15%, ${rgba('#ffffff', 0.5)} 38%, transparent 47%, ${rgba(p.ink, 0.12)} 62%, transparent 78%)`,
   199→      backgroundSize: 'auto',
   200→    }
   201→  }
   202→  if (/gloss|lacquer|glaze|polish|enamel|patent|vinyl|glass|crystal|iridescent|holograph/.test(t)) {
   203→    return {
   204→      backgroundImage: `linear-gradient(115deg, ${rgba('#ffffff', 0.4)} 0%, transparent 30%), linear-gradient(335deg, ${rgba(p.accent, 0.18)} 0%, transparent 40%)`,
   205→      backgroundSize: 'auto',
   206→    }
   207→  }
   208→  if (/knit|stitch|embroider|crochet|mesh|net|perforat|grid|lattice|sashiko|kantha/.test(t)) {
   209→    return {
   210→      backgroundImage: `radial-gradient(${inkA} 1px, transparent 1.2px)`,
   211→      backgroundSize: '6px 6px',
   212→    }
   213→  }
   214→  if (/paper|card|newsprint|parchment|grain|kraft|dust|sand|foxing|noise|halftone/.test(t)) {
   215→    return {
   216→      backgroundImage: `radial-gradient(${rgba(p.ink, 0.07)} 0.6px, transparent 0.7px)`,
   217→      backgroundSize: '3px 3px',
   218→    }
   219→  }
   220→  if (/rust|corrode|tarnish|weather|distress|peel|crack|craze|aged|faded|worn|patina/.test(t)) {
   221→    return {
   222→      backgroundImage: `radial-gradient(ellipse at 20% 30%, ${rgba(p.accent, 0.2)}, transparent 45%), radial-gradient(ellipse at 80% 70%, ${rgba(p.ink, 0.14)}, transparent 50%), radial-gradient(${inkSoft} 0.7px, transparent 0.8px)`,
   223→      backgroundSize: 'auto, auto, 9px 9px',
   224→    }
   225→  }
   226→  if (/moss|bark|organic|botanic|leaf|coral|shell|nacre|feather|lichen/.test(t)) {
   227→    return {
   228→      backgroundImage: `radial-gradient(ellipse at 25% 60%, ${rgba(p.support, 0.25)}, transparent 50%), radial-gradient(ellipse at 75% 25%, ${rgba(p.accent, 0.16)}, transparent 45%)`,
   229→      backgroundSize: 'auto',
   230→    }
   231→  }
   232→  // default: subtle grain
   233→  return {
   234→    backgroundImage: `radial-gradient(${rgba(p.ink, 0.05)} 0.6px, transparent 0.7px)`,
   235→    backgroundSize: '4px 4px',
   236→  }

   243→/* ------------------------------ lighting & motion ------------------------------ */
   244→
   245→function lightingOverlay(a: AestheticFull, p: DemoPalette): string | null {
   246→  const lit = `${a.lighting.temperature ?? ''} ${a.lighting.quality ?? ''} ${a.lighting.shadow ?? ''}`.toLowerCase()
   247→  if (!lit.trim()) return null
   248→  const warm = /warm|golden|candle|amber|tungsten|sunset/.test(lit)
   249→  const cool = /cool|blue|moon|neon|fluoresc|cold|cyan|icy/.test(lit)
   250→  if (warm && !cool) return `linear-gradient(180deg, rgba(255, 176, 80, 0.10), transparent 70%)`
   251→  if (cool && !warm) return `linear-gradient(180deg, rgba(96, 140, 200, 0.10), transparent 70%)`
   252→  return null
   253→}
   254→
   255→function shadowFor(a: AestheticFull): string {
   256→  const q = `${a.lighting.quality ?? ''} ${a.lighting.shadow ?? ''}`.toLowerCase()
   257→  if (/hard|sharp|harsh|crisp|noir|dramatic|high contrast/.test(q)) return '0 10px 28px -10px rgba(28,25,23,0.5)'
   258→  if (/soft|diffuse|gentle|even|flat/.test(q)) return '0 6px 20px -8px rgba(28,25,23,0.18)'
   259→  return '0 8px 24px -10px rgba(28,25,23,0.28)'
   260→}

   262→type MotionKind = 'float' | 'flicker' | 'none'
   263→
   264→function motionFor(a: AestheticFull): MotionKind {
   265→  const m = `${a.uiTranslation.motion ?? ''} ${a.category}`.toLowerCase()
   266→  if (/glitch|flicker|neon|strobe|vhs|scan|interference|crt|blink/.test(m)) return 'flicker'
   267→  if (/slow|gentle|float|parallax|glide|drift|hover|smooth|playful|bounce|spring/.test(m)) return 'float'
   268→  return 'none'
   269→}

   271→/* ------------------------------ hash (deterministic) ------------------------------ */
   272→
   273→function hashStr(s: string): number {
   274→  let h = 2166136261
   275→  for (let i = 0; i < s.length; i++) {
   276→    h ^= s.charCodeAt(i)
   277→    h = Math.imul(h, 16777619)
   278→  }
   279→  return Math.abs(h)
   280→}

   288→function firstSentence(text: string, max = 130): string {
   289→  const s = text.split(/(?<=[.!?])\s/)[0] ?? text
   290→  return s.length > max ? s.slice(0, max - 1).trimEnd() + '…' : s
   291→}

   293→export function StyleDemo({ a }: { a: AestheticFull }) {
   294→  const p = useMemo(() => derivePalette(a.colors), [a.colors])

   682→export function VisualGallery({ images, name }: { images: ImageEntry[]; name: string }) {
   683→  const [failed, setFailed] = useState<Set<string>>(() => new Set())
   684→  const [zoomed, setZoomed] = useState<string | null>(null)
   685→  const visible = images.filter((i) => !failed.has(i.url))
   686→
   687→  // Escape closes the lightbox before the enclosing sheet.
   688→  useEffect(() => {
   689→    if (!zoomed) return
   690→    const onKey = (e: KeyboardEvent) => {
   691→      if (e.key === 'Escape') {
   692→        e.stopImmediatePropagation()
   693→        setZoomed(null)
   694→      }
   695→    }
   696→    window.addEventListener('keydown', onKey, true)
   697→    return () => window.removeEventListener('keydown', onKey, true)
   698→  }, [zoomed])
   699→  if (visible.length === 0) return null
   700→
   701→  return (
   702→    <section aria-label="Visual examples">
   703→      <div className="mb-3 flex items-center justify-between gap-3">
   704→        <div>
   705→          <h3 className="flex items-center gap-2 font-serif text-lg text-stone-900">

   707→            Visual examples

   712→        </div>
   713→      </div>
