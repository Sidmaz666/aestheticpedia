    54→    54→/** Per-entry record completeness — part of the no-missing-data audit policy. */
    55→    55→function RecordMeter({ a }: { a: AestheticFull }) {
    56→    56→  const checks: Array<{ label: string; ok: boolean }> = [
    57→    57→    { label: 'palette', ok: a.colors.length > 0 },
    58→    58→    { label: 'period', ok: Boolean(a.periodStart || a.era) },
    59→    59→    { label: 'origin', ok: Boolean(a.origin || a.geography) },
    60→    60→    { label: 'materials', ok: a.materials.length > 0 },
    61→    61→    { label: 'textures', ok: a.textures.length > 0 },
    62→    62→    { label: 'objects', ok: a.objects.length > 0 },
    63→    63→    { label: 'examples', ok: a.keyExamples.length > 0 },
    64→    64→    { label: 'visual DNA', ok: Object.keys(a.visualDNA).length > 0 },
    65→    65→    { label: 'typography', ok: Object.keys(a.typography).length > 0 },
    66→    66→    { label: 'lighting', ok: Object.keys(a.lighting).length > 0 },
    67→    67→    { label: 'UI translation', ok: Object.keys(a.uiTranslation).length > 0 },
    68→    68→    { label: 'recipe', ok: Object.keys(a.recipe).length > 0 },
    69→    69→    { label: 'sonic identity', ok: a.sounds.length > 0 },
    70→    70→    { label: 'sources', ok: a.sources.length > 0 },
    71→    71→    { label: 'images', ok: a.images.length > 0 },
    72→    72→  ]
    73→    73→  const ok = checks.filter((c) => c.ok).length
    74→    74→  const missing = checks.filter((c) => !c.ok).map((c) => c.label)
    75→    75→  const pct = Math.round((ok / checks.length) * 100)
    76→    76→  return (
    77→    77→    <section
    78→    78→      className="rounded-lg border border-stone-200 bg-stone-50 px-4 py-3"
    79→    79→      aria-label="Record completeness"
    80→    80→    >
    81→    81→      <div className="flex items-center justify-between gap-3 text-sm">
    82→    82→        <span className="font-medium text-stone-700">
    83→    83→          Record completeness — {ok}/{checks.length} fields documented
    84→    84→        </span>
    85→    85→        <span className={`tabular-nums ${pct >= 90 ? 'text-emerald-800' : pct >= 60 ? 'text-amber-800' : 'text-stone-500'}`}>
    86→    86→          {pct}%
    87→    87→        </span>
    88→    88→      </div>
    89→    89→      <Progress value={pct} className="mt-2 h-1.5" />
    90→    90→      <p className="mt-1.5 text-xs text-stone-500">
    91→    91→        {missing.length === 0
    92→    92→          ? 'Every documented field is populated for this entry.'
    93→    93→          : `Still documenting: ${missing.join(', ')}. The research pipeline fills gaps continuously, most-visited entries first.`}
    94→    94→      </p>
    95→    95→    </section>
    96→    96→  )
    97→    97→}

   662→   662→  note?: string

   664→   664→  source?: { slug: string; name: string; category: string }

   667→   667→function RelationsSection({
   668→   668→  relations,
   669→   669→  onNavigate,
   670→   670→}: {
   671→   671→  relations: { outgoing: RelationEdge[]; incoming: RelationEdge[] }
   672→   672→  onNavigate: (slug: string) => void
   673→   673→}) {
   674→   674→  if (relations.outgoing.length === 0 && relations.incoming.length === 0) {
   675→   675→    return (
   676→   676→      <section aria-label="Relationships">
   677→   677→        <SectionHeading>Relationships</SectionHeading>
   678→   678→        <p className="text-sm text-stone-500">No documented relationships yet — the pipeline may link this entry later.</p>
   679→   679→      </section>
   680→   680→    )
   681→   681→  }
   682→   682→
   683→   683→  const group = (edges: RelationEdge[]) => {
   684→   684→    const map = new Map<string, RelationEdge[]>()
   685→   685→    for (const e of edges) {
   686→   686→      const list = map.get(e.type) ?? []
   687→   687→      list.push(e)
   688→   688→      map.set(e.type, list)
   689→   689→    }
   690→   690→    return [...map.entries()]
   691→   691→  }
   692→   692→
   693→   693→  return (
   694→   694→    <section aria-label="Relationships">
   695→   695→      <SectionHeading hint="The aesthetic family tree.">Relationships</SectionHeading>
   696→   696→      <div className="space-y-4">
   697→   697→        {relations.outgoing.length > 0 && (
   698→   698→          <div>
   699→   699→            <p className="mb-1.5 flex items-center gap-1.5 text-[11px] font-medium uppercase tracking-[0.14em] text-stone-500">
   700→   700→              <ArrowRight className="h-3 w-3" aria-hidden="true" /> Outgoing
   701→   701→            </p>
   702→   702→            <div className="space-y-2.5">
   703→   703→              {group(relations.outgoing).map(([type, edges]) => (
   704→   704→                <div key={type} className="flex flex-wrap items-center gap-1.5">
   705→   705→                  <span className="mr-1 min-w-fit text-xs capitalize text-stone-400">
   706→   706→                    {labelize(type)}:
   707→   707→                  </span>
   708→   708→                  {edges.map((e) =>
   709→   709→                    e.target ? (
   710→   710→                      <RelationChip
   711→   711→                        key={`${e.target.slug}-${type}`}
   712→   712→                        name={e.target.name}
   713→   713→                        note={e.note}
   714→   714→                        onClick={() => onNavigate(e.target!.slug)}
   715→   715→                      />
   716→   716→                    ) : null
   717→   717→                  )}
   718→   718→                </div>
   719→   719→              ))}
   720→   720→            </div>
   721→   721→          </div>
   722→   722→        )}
   723→   723→        {relations.incoming.length > 0 && (
   724→   724→          <div>
   725→   725→            <p className="mb-1.5 flex items-center gap-1.5 text-[11px] font-medium uppercase tracking-[0.14em] text-stone-500">
   726→   726→              <ArrowLeft className="h-3 w-3" aria-hidden="true" /> Incoming
   727→   727→            </p>
   728→   728→            <div className="space-y-2.5">
   729→   729→              {group(relations.incoming).map(([type, edges]) => (
   730→   730→                <div key={type} className="flex flex-wrap items-center gap-1.5">
   731→   731→                  <span className="mr-1 min-w-fit text-xs capitalize text-stone-400">
   732→   732→                    {labelize(type)}:
   733→   733→                  </span>
   734→   734→                  {edges.map((e) =>
   735→   735→                    e.source ? (
   736→   736→                      <RelationChip

   738→   738→                        name={e.source.name}
   739→   739→                        note={e.note}

   741→   741→                      />
