    25→    25→function hexToRgb(hex: string): [number, number, number] | null {
    26→    26→  let h = hex.replace('#', '').trim()
    27→    27→  if (h.length === 3) h = h.split('').map((c) => c + c).join('')
    28→    28→  if (!/^[0-9a-fA-F]{6}$/.test(h)) return null
    29→    29→  return [parseInt(h.slice(0, 2), 16), parseInt(h.slice(2, 4), 16), parseInt(h.slice(4, 6), 16)]
    30→    30→}
    31→    31→
    32→    32→/** Relative luminance 0 (dark) – 1 (light). */
    33→    33→function luminance(hex: string): number {
    34→    34→  const rgb = hexToRgb(hex)
    35→    35→  if (!rgb) return 0.5
    36→    36→  const [r, g, b] = rgb.map((v) => {
    37→    37→    const s = v / 255
    38→    38→    return s <= 0.03928 ? s / 12.92 : Math.pow((s + 0.055) / 1.055, 2.4)
    39→    39→  })
    40→    40→  return 0.2126 * r + 0.7152 * g + 0.0722 * b
    41→    41→}
    42→    42→
    43→    43→function saturation(hex: string): number {
    44→    44→  const rgb = hexToRgb(hex)
    45→    45→  if (!rgb) return 0
    46→    46→  const [r, g, b] = rgb.map((v) => v / 255)
    47→    47→  const max = Math.max(r, g, b)
    48→    48→  const min = Math.min(r, g, b)
    49→    49→  if (max === min) return 0
    50→    50→  const l = (max + min) / 2
    51→    51→  return l > 0.5 ? (max - min) / (2 - max - min) : (max - min) / (max + min)
    52→    52→}
    53→    53→
    54→    54→function rgba(hex: string, alpha: number): string {
    55→    55→  const rgb = hexToRgb(hex) ?? [137, 109, 59]
    56→    56→  return `rgba(${rgb[0]}, ${rgb[1]}, ${rgb[2]}, ${alpha})`
    57→    57→}
    58→    58→
    59→    59→/** Black or white text color for a given background. */
    60→    60→function readableOn(hex: string): string {
    61→    61→  return luminance(hex) > 0.45 ? '#1c1917' : '#fdfcf8'
    62→    62→}

    84→    84→/** Derive demo roles (bg/surface/ink/accent) from the entry palette. */
    85→    85→function derivePalette(colors: ColorEntry[]): DemoPalette {
    86→    86→  const valid = colors.filter((c) => hexToRgb(c.hex))
    87→    87→  if (valid.length < 2) return FALLBACK
    88→    88→  const sorted = [...valid].sort((a, b) => luminance(b.hex) - luminance(a.hex))
    89→    89→  const bg = sorted[0].hex
    90→    90→  const ink = sorted[sorted.length - 1].hex
    91→    91→  // accent: most saturated color that is neither bg nor ink
    92→    92→  const mid = valid.filter((c) => c.hex !== bg && c.hex !== ink)
    93→    93→  const accent =
    94→    94→    mid.length > 0
    95→    95→      ? [...mid].sort((a, b) => saturation(b.hex) * (1 - Math.abs(luminance(b.hex) - 0.5)) - saturation(a.hex) * (1 - Math.abs(luminance(a.hex) - 0.5)))[0].hex
    96→    96→      : sorted[Math.floor(sorted.length / 2)].hex
    97→    97→  const surface = sorted.length > 2 ? sorted[1].hex : mix(bg, ink, 0.08)
    98→    98→  const support = mid.find((c) => c.hex !== accent)?.hex ?? mix(accent, bg, 0.3)
    99→    99→  const warm = isWarm(accent)
   100→   100→  return { bg, surface, ink, accent, support, warm, light: luminance(bg) > 0.5 }
   101→   101→}
   102→   102→
   103→   103→function mix(a: string, b: string, t: number): string {
   104→   104→  const ca = hexToRgb(a)
   105→   105→  const cb = hexToRgb(b)
   106→   106→  if (!ca || !cb) return a
   107→   107→  const c = ca.map((v, i) => Math.round(v + (cb[i] - v) * t))
   108→   108→  return `#${c.map((v) => v.toString(16).padStart(2, '0')).join('')}`
   109→   109→}
   110→   110→
   111→   111→function isWarm(hex: string): boolean {
   112→   112→  const rgb = hexToRgb(hex)
   113→   113→  if (!rgb) return true
   114→   114→  const [r, g, b] = rgb
   115→   115→  return r + r * 0.2 + g * 0.3 >= b
   116→   116→}

   125→   125→function fontFor(text: string, fallback: 'serif' | 'sans'): string {
   126→   126→  const t = (text || '').toLowerCase()
   127→   127→  if (/mono|typewriter|courier|terminal/.test(t)) return MONO
   128→   128→  if (/script|cursive|hand|brush letter|calligrap/.test(t)) return SCRIPT
   129→   129→  if (/serif|didone|garamond|roman|slab|bodoni|clarendon|garamond/.test(t) && !/sans/.test(t)) return SERIF
   130→   130→  if (/sans|grotesque|geometric|gothic|helvetica|futura|swiss|arial|neo-grotesque|rounded/.test(t)) return SANS
   131→   131→  if (fallback === 'serif') return SERIF
   132→   132→  return SANS
   133→   133→}

   150→   150→    transform: deco || /all caps|uppercase|poster/.test(t) ? 'uppercase' : 'none',
   151→   151→    weight: mono ? 500 : 700,

   245→   245→function lightingOverlay(a: AestheticFull, p: DemoPalette): string | null {
   246→   246→  const lit = `${a.lighting.temperature ?? ''} ${a.lighting.quality ?? ''} ${a.lighting.shadow ?? ''}`.toLowerCase()
   247→   247→  if (!lit.trim()) return null
   248→   248→  const warm = /warm|golden|candle|amber|tungsten|sunset/.test(lit)
   249→   249→  const cool = /cool|blue|moon|neon|fluoresc|cold|cyan|icy/.test(lit)
   250→   250→  if (warm && !cool) return `linear-gradient(180deg, rgba(255, 176, 80, 0.10), transparent 70%)`
   251→   251→  if (cool && !warm) return `linear-gradient(180deg, rgba(96, 140, 200, 0.10), transparent 70%)`
   252→   252→  return null
   253→   253→}

   255→   255→function shadowFor(a: AestheticFull): string {
   256→   256→  const q = `${a.lighting.quality ?? ''} ${a.lighting.shadow ?? ''}`.toLowerCase()
   257→   257→  if (/hard|sharp|harsh|crisp|noir|dramatic|high contrast/.test(q)) return '0 10px 28px -10px rgba(28,25,23,0.5)'
   258→   258→  if (/soft|diffuse|gentle|even|flat/.test(q)) return '0 6px 20px -8px rgba(28,25,23,0.18)'
   259→   259→  return '0 8px 24px -10px rgba(28,25,23,0.28)'
   260→   260→}

   272→   272→
   273→   273→function hashStr(s: string): number {
   274→   274→  let h = 2166136261
   275→   275→  for (let i = 0; i < s.length; i++) {
   276→   276→    h ^= s.charCodeAt(i)
   277→   277→    h = Math.imul(h, 16777619)
   278→   278→  }
   279→   279→  return Math.abs(h)
   280→   280→}

   698→   698→  }, [zoomed])
   699→   699→  if (visible.length === 0) return null
   700→   700→

   702→   702→    <section aria-label="Visual examples">
   703→   703→      <div className="mb-3 flex items-center justify-between gap-3">
